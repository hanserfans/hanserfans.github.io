## 1. 条件语句

### 1.1 基本的if语句

```python
age = 18

if age >= 18:
    print("你已经成年了！")
    print("可以投票了！")
```

### 1.2 if-else语句

```python
score = 75

if score >= 60:
    print("🎉 恭喜你，及格了！")
else:
    print("😢 很遗憾，你没有及格。")
    print("继续努力！")
```

### 1.3 if-elif-else语句

```python
score = 85

if score >= 90:
    grade = "A"
    description = "优秀"
elif score >= 80:
    grade = "B"
    description = "良好"
elif score >= 70:
    grade = "C"
    description = "中等"
elif score >= 60:
    grade = "D"
    description = "及格"
else:
    grade = "F"
    description = "不及格"

print(f"得分: {score}")
print(f"等级: {grade} - {description}")
```

### 1.4 嵌套条件

```python
age = 20
has_license = True

if age >= 18:
    print("年龄符合要求")
    if has_license:
        print("✅ 你可以租车")
    else:
        print("❌ 需要先考驾照")
else:
    print("年龄不符合要求")
    print("需要等到18岁")
```

### 1.5 逻辑运算符

```python
# and: 多个条件都需要满足
age = 25
income = 50000

if age >= 18 and income >= 30000:
    print("符合贷款条件")

# or: 满足任一条件即可
has_card = False
has_cash = True

if has_card or has_cash:
    print("可以支付")

# not: 取反
is_blocked = False

if not is_blocked:
    print("账户正常")
```

### 1.6 三元运算符

```python
age = 20

# 传统写法
if age >= 18:
    status = "成年人"
else:
    status = "未成年人"

# 三元运算符（简洁写法）
status = "成年人" if age >= 18 else "未成年人"

print(f"状态: {status}")
```

## 2. 循环结构

### 2.1 for循环

```python
# 遍历列表
fruits = ["苹果", "香蕉", "橙子", "葡萄"]

print("🍎 水果列表:")
for fruit in fruits:
    print(f"  - {fruit}")

# 遍历字符串
word = "Python"
print("\n📝 遍历字符串:")
for char in word:
    print(f"  {char}")

# 使用range()
print("\n🔢 数字序列:")
for i in range(5):
    print(f"  {i}")

# range(start, stop, step)
print("\n📊 偶数序列:")
for i in range(0, 10, 2):
    print(f"  {i}")

# 遍历字典
person = {"name": "张三", "age": 25, "city": "北京"}
print("\n👤 字典遍历:")
for key in person:
    print(f"  {key}: {person[key]}")
```

### 2.2 while循环

```python
# 基本的while循环
count = 0

print("🔢 while循环计数:")
while count < 5:
    print(f"  count = {count}")
    count += 1

print("循环结束!")

# while与条件结合
print("\n🎮 猜数字游戏:")
secret_number = 7
guess = 0
attempts = 0

while guess != secret_number:
    guess = int(input("请输入你猜的数字: "))
    attempts += 1
    
    if guess < secret_number:
        print("太小了，再试一次！")
    elif guess > secret_number:
        print("太大了，再试一次！")

print(f"🎉 恭喜你！猜对了！")
print(f"一共猜了 {attempts} 次")
```

### 2.3 循环嵌套

```python
# 打印九九乘法表
print("📊 九九乘法表:")
print("-" * 50)

for i in range(1, 10):
    row = ""
    for j in range(1, i + 1):
        result = i * j
        row += f"{j}×{i}={result:2d}  "
    print(row)

# 打印三角形
print("\n🔺 三角形图案:")
n = 5

for i in range(1, n + 1):
    # 打印空格
    for j in range(n - i):
        print(" ", end="")
    # 打印星号
    for k in range(2 * i - 1):
        print("*", end="")
    print()
```

## 3. break和continue

### 3.1 break语句

`break` 用于**跳出**整个循环：

```python
print("🔍 查找第一个大于50的数:")
numbers = [12, 25, 37, 48, 63, 72, 89]

for num in numbers:
    print(f"检查: {num}")
    if num > 50:
        print(f"✅ 找到第一个大于50的数: {num}")
        break  # 找到后立即跳出循环
```

### 3.2 continue语句

`continue` 用于**跳过**当前迭代，继续下一次循环：

```python
print("📊 统计不及格人数:")
scores = [85, 45, 72, 55, 38, 90, 62]

failed_count = 0
for i, score in enumerate(scores):
    if score >= 60:
        continue  # 及格，跳过
    failed_count += 1
    print(f"  第{i+1}个学生: {score}分 - 不及格")

print(f"\n不及格人数: {failed_count}")
```

### 3.3 else子句

循环的`else`子句在循环**正常完成**时执行（不是break跳出时）：

```python
print("📝 查找元素示例:")

numbers = [1, 3, 5, 7, 9]
target = 6

for num in numbers:
    if num == target:
        print(f"✅ 找到目标: {target}")
        break
else:
    # 如果循环没有被break终止
    print(f"❌ 未找到目标: {target}")
```

## 4. 函数基础

### 4.1 定义函数

```python
# 无参数函数
def greet():
    """简单的问候函数"""
    print("你好！欢迎学习Python！")

# 调用函数
greet()
greet()

def greet_person(name):

```

## 5. 函数的参数

### 5.1 位置参数

```python
def introduce(name, age, city):
    """位置参数示例"""
    print(f"我叫{name}，今年{age}岁，来自{city}。")

# 按顺序传入参数
introduce("张三", 25, "北京")

# 使用关键字参数（顺序可以改变）
introduce(city="上海", age=30, name="李四")
```

### 5.2 默认参数

```python
def create_profile(name, age=18, city="未知", occupation="学生"):
    """带默认参数的函数"""
    print("=" * 40)
    print(f"姓名: {name}")
    print(f"年龄: {age}")
    print(f"城市: {city}")
    print(f"职业: {occupation}")
    print("=" * 40)

# 使用默认参数
create_profile("小明")
print()
create_profile("小红", age=22, city="广州")
```

### 5.3 可变参数 (*args)

```python
def sum_all(*numbers):
    """计算所有参数的和"""
    total = 0
    print(f"接收到 {len(numbers)} 个数字: {numbers}")
    
    for num in numbers:
        total += num
    
    return total

result = sum_all(1, 2, 3)
print(f"1+2+3 = {result}")
print()

result = sum_all(10, 20, 30, 40, 50)
print(f"10+20+30+40+50 = {result}")
```

### 5.4 关键字参数 (**kwargs)

```python
def print_info(**info):
    """打印所有关键字参数"""
    print("\n📋 信息汇总:")
    print("-" * 40)
    
    for key, value in info.items():
        print(f"  {key}: {value}")

print_info(name="张三", age=25, city="北京", job="工程师")
print()
print_info(phone="13800138000", email="zhangsan@example.com")
```

### 5.5 参数组合

```python
def complex_function(name, age=18, *args, city="未知", **kwargs):
    """参数组合示例"""
    print(f"姓名: {name}")
    print(f"年龄: {age}")
    print(f"城市: {city}")
    
    if args:
        print(f"其他位置参数: {args}")
    
    if kwargs:
        print(f"其他关键字参数: {kwargs}")

complex_function("小明", 20, "额外1", "额外2", city="深圳", job="开发")
```

## 6. 函数的返回值

### 6.1 单个返回值

```python
def calculate(a, b):
    """返回多个计算结果"""
    return a + b

result = calculate(10, 5)
print(f"结果: {result}")
```

### 6.2 多个返回值

```python
def calculate(a, b):
    """返回多个计算结果"""
    sum_result = a + b
    difference = a - b
    product = a * b
    
    if b != 0:
        quotient = a / b
    else:
        quotient = None
    
    return sum_result, difference, product, quotient

# 接收多个返回值
s, d, p, q = calculate(20, 4)

print("=" * 40)
print(f"加法: 20 + 4 = {s}")
print(f"减法: 20 - 4 = {d}")
print(f"乘法: 20 × 4 = {p}")
print(f"除法: 20 ÷ 4 = {q}")
```

### 6.3 提前返回

```python
def divide(a, b):
    """除法运算，带错误检查"""
    if b == 0:
        print("❌ 错误：除数不能为零！")
        return None  # 提前返回
    
    return a / b

print(f"10 ÷ 2 = {divide(10, 2)}")
print(f"10 ÷ 0 = {divide(10, 0)}")
```

## 7. 变量作用域

### 7.1 局部变量和全局变量

```python
# 全局变量
global_var = "我是全局变量"

def test_scope():
    # 局部变量
    local_var = "我是局部变量"
    print(f"函数内访问局部变量: {local_var}")
    print(f"函数内访问全局变量: {global_var}")

test_scope()
# print(local_var)  # ❌ 错误：局部变量不能在函数外访问
print(f"函数外访问全局变量: {global_var}")
```

### 7.2 global关键字

```python
counter = 0  # 全局变量

def increment():
    """使用global关键字修改全局变量"""
    global counter
    counter += 1
    print(f"计数器值: {counter}")

print(f"初始值: {counter}")
increment()
increment()
increment()
print(f"最终值: {counter}")
```

### 7.3 nonlocal关键字

```python
def outer():
    """外部函数"""
    count = 0
    
    def inner():
        """内部函数"""
        nonlocal count  # 使用nonlocal访问外部函数的变量
        count += 1
        print(f"内部函数计数: {count}")
    
    inner()
    inner()
    print(f"外部函数计数: {count}")

outer()
```

## 8. lambda表达式

### 8.1 基本语法

```python
# 普通函数
def square(x):
    return x ** 2

# lambda表达式
square_lambda = lambda x: x ** 2

print(f"square(5) = {square(5)}")
print(f"square_lambda(5) = {square_lambda(5)}")
```

### 8.2 带参数的lambda

```python
# 两个参数
add = lambda a, b: a + b
print(f"add(3, 5) = {add(3, 5)}")

# 带条件
max_value = lambda a, b: a if a > b else b
print(f"max(10, 20) = {max_value(10, 20)}")

# 带默认值
greet = lambda name, greeting="你好": f"{greeting}，{name}！"
print(greet("小明"))
print(greet("小红", "早上好"))
```

### 8.3 lambda与内置函数结合

```python
# 与map()结合
numbers = [1, 2, 3, 4, 5]
squares = list(map(lambda x: x ** 2, numbers))
print(f"平方: {squares}")

# 与filter()结合
even_numbers = list(filter(lambda x: x % 2 == 0, numbers))
print(f"偶数: {even_numbers}")

# 与sorted()结合
students = [
    {"name": "张三", "score": 85},
    {"name": "李四", "score": 92},
    {"name": "王五", "score": 78}
]

sorted_students = sorted(students, key=lambda x: x["score"], reverse=True)
print("按成绩排序:")
for s in sorted_students:
    print(f"  {s['name']}: {s['score']}")

# 与reduce()结合
from functools import reduce

total = reduce(lambda x, y: x + y, numbers)
print(f"总和: {total}")
```

### 练习1：斐波那契数列

📝 **要求**：使用循环实现斐波那契数列

```python
def fibonacci(n):
    """生成斐波那契数列的前n项"""
    if n <= 0:
        return []
    elif n == 1:
        return [0]
    
    fib = [0, 1]
    while len(fib) < n:
        fib.append(fib[-1] + fib[-2])
    
    return fib

# 测试
n = 10
result = fibonacci(n)
print(f"斐波那契数列前{n}项:")
print(" ".join(map(str, result)))
```

### 练习2：质数判断

📝 **要求**：编写函数判断一个数是否为质数

```python
def is_prime(num):
    """判断是否为质数"""
    if num < 2:
        return False
    if num == 2:
        return True
    if num % 2 == 0:
        return False
    
    for i in range(3, int(num ** 0.5) + 1, 2):
        if num % i == 0:
            return False
    
    return True

# 找出100以内的所有质数
print("100以内的质数:")
primes = [i for i in range(2, 101) if is_prime(i)]
print(primes)
```

### 练习3：汉诺塔问题

📝 **要求**：实现汉诺塔的移动步骤

```python
def hanoi(n, source, target, auxiliary):
    """汉诺塔问题的递归解法"""
    if n == 1:
        print(f"将盘子从 {source} 移动到 {target}")
        return
    
    # 将n-1个盘子从源柱子移动到辅助柱子
    hanoi(n - 1, source, auxiliary, target)
    
    # 将最大的盘子从源柱子移动到目标柱子
    print(f"将盘子从 {source} 移动到 {target}")
    
    # 将n-1个盘子从辅助柱子移动到目标柱子
    hanoi(n - 1, auxiliary, target, source)

# 测试3个盘子
print("=" * 40)
print("汉诺塔解法（3个盘子）:")
hanoi(3, 'A', 'C', 'B')
```

### 练习4：实现简单的计算器

📝 **要求**：使用函数实现一个计算器

```python
def calculator(a, b, operator):
    """简单计算器"""
    operations = {
        '+': lambda x, y: x + y,
        '-': lambda x, y: x - y,
        '*': lambda x, y: x * y,
        '/': lambda x, y: x / y if y != 0 else "错误：除数为零"
    }
    
    if operator not in operations:
        return "错误：无效的操作符"
    
    return operations[operator](a, b)

# 测试
print("🧮 计算器测试:")
print(f"10 + 5 = {calculator(10, 5, '+')}")
print(f"10 - 5 = {calculator(10, 5, '-')}")
print(f"10 × 5 = {calculator(10, 5, '*')}")
print(f"10 ÷ 5 = {calculator(10, 5, '/')}")
print(f"10 ÷ 0 = {calculator(10, 0, '/')}")
```
